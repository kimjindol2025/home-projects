module kernel_core {
    // ----------------------------
    // Kernel Core Structures
    // ----------------------------
    type Task {
        id: int
        name: string
        state: string      // READY, RUNNING, WAITING, TERMINATED
        priority: int
        stack: list
        context: map
    }

    var task_table: map[int, Task] = {}
    var ready_queue: list[int] = []

    // ----------------------------
    // Task Management
    // ----------------------------
    func create_task(id: int, name: string, priority: int, init_context: map) -> Task {
        var t = Task{id=id, name=name, state="READY", priority=priority, stack=[], context=init_context}
        task_table[id] = t
        ready_queue.push(id)
        return t
    }

    func terminate_task(id: int) {
        if task_table.contains(id) {
            task_table[id].state = "TERMINATED"
            ready_queue = ready_queue.filter(x -> x != id)
        }
    }

    // ----------------------------
    // Task Scheduler (Simple Priority Round-Robin)
    // ----------------------------
    func schedule() {
        if ready_queue.length == 0 { return }

        // 높은 우선순위 우선
        ready_queue.sort((a,b) -> task_table[b].priority - task_table[a].priority)

        var next_id = ready_queue[0]
        var t = task_table[next_id]

        if t.state == "READY" {
            t.state = "RUNNING"
            // 여기서 실제 Task 실행 코드 삽입 가능
            // 예: run_task_logic(t)
            t.state = "WAITING" // 실행 후 상태 변경
        }

        // Round-Robin: 맨 뒤로 이동
        ready_queue.push(ready_queue.shift())
    }

    // ----------------------------
    // Task Debug / Info
    // ----------------------------
    func list_tasks() {
        for id, t in task_table {
            print("Task:", t.id, t.name, "State:", t.state, "Priority:", t.priority)
        }
    }
}

이 코드는:

Task 구조체: 상태, 우선순위, 스택, 컨텍스트 포함

Task Table + Ready Queue: 커널 내부 관리

Scheduler: 단순 Priority Round-Robin 방식

Task 생성/종료/조회 함수 포함
