1. tree-to-json.fl 설계 스펙
이 모듈은 재귀적 탐색을 통해 파일 시스템의 하이러키를 추출하며, 각 노드에 대한 메타데이터를 포함합니다.
# tree-to-json.fl (Stage 1-1 Core)

fn buildWebNode(hash: string, name: string, type: string): map {
    let node = {
        "id": hash,           # 고유 식별자 (FreeLang Hash)
        "name": name,         # 파일/폴더 이름
        "type": type,         # 'blob' (파일) 또는 'tree' (폴더)
        "children": []        # 하위 노드 리스트
    }
    
    if type == "tree" {
        let entries = readTreeObject(hash) # Stage 1 엔진 호출
        for entry in entries {
            node["children"].append(
                buildWebNode(entry.hash, entry.name, entry.type)
            )
        }
    }
    return node
}
