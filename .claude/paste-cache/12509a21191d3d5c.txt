is.pos = 0
    this.currentChar = source[0] || null
  }

  advance() {
    this.pos++
    this.currentChar = this.pos < this.source.length ? this.source[this.pos] : null
  }

  skipWhitespace() {
    while (this.currentChar && /\s/.test(this.currentChar)) this.advance()
  }

  identifier() {
    let result = ''
    while (this.currentChar && /[a-zA-Z0-9_]/.test(this.currentChar)) {
      result += this.currentChar
      this.advance()
    }
    return { type: 'IDENTIFIER', value: result }
  }

  number() {
    let result = ''
    while (this.currentChar && /[0-9]/.test(this.currentChar)) {
      result += this.currentChar
      this.advance()
    }
    return { type: 'NUMBER', value: Number(result) }
  }

  getNextToken() {
    this.skipWhitespace()
    if (!this.currentChar) return { type: 'EOF' }

    if (/[a-zA-Z_]/.test(this.currentChar)) return this.identifier()
    if (/[0-9]/.test(this.currentChar)) return this.number()

    const char = this.currentChar
    this.advance()
    return { type: char, value: char }
  }

  tokenize() {
    const tokens = []
    let tok
    do {
      tok = this.getNextToken()
      tokens.push(tok)
    } while (tok.type !== 'EOF')
    return tokens
  }
}

// parser.js
import { Lexer } from './lexer.js'

export class Parser {
  constructor(source) {
    this.lexer = new Lexer(source)
    this.tokens = this.lexer.tokenize()
    this.pos = 0
    this.currentToken = this.tokens[this.pos]
  }

  eat(type) {
    if (this.currentToken.type === type) this.currentToken = this.tokens[++this.pos]
    else throw new SyntaxError(`Expected ${type} but got ${this.currentToken.type}`)
  }

  factor() {
    const tok = this.currentToken
    if (tok.type === 'NUMBER') {
      this.eat('NUMBER')
      return { type: 'Literal', value: tok.value }
    } else if (tok.type === 'IDENTIFIER') {
      this.eat('IDENTIFIER')
      return { type: 'Variable', name: tok.value }
    } else if (tok.type === '(') {
      this.eat('(')
      const node = this.expr()
      this.eat(')')
      return node
    }
    throw new SyntaxError(`Unexpected token: ${tok.type}`)
  }

  term() {
    let node = this.factor()
    while (['*', '/'].includes(this.currentToken.type)) {
      const op = this.currentToken.type
      this.eat(op)
      node = { type: 'BinaryExpression', operator: op, left: node, right: this.factor() }
    }
    return node
  }

  expr() {
    let node = this.term()
    while (['+', '-'].includes(this.currentToken.type)) {
      const op = this.currentToken.type
      this.eat(op)
      node = { type: 'BinaryExpression', operator: op, left: node, right: this.term() }
    }
    return node
  }

  parse() {
    const ast = this.expr()
    if (this.currentToken.type !== 'EOF') throw new SyntaxError('Unexpected end of input')
    return ast
  }
}

// semantic.js
export class SemanticAnalyzer {
  constructor() {
    this.symbols = new Map()
  }

  declareVariable(name, type) {
    if (this.symbols.has(name)) throw new Error(`Variable ${name} already declared`)
    this.symbols.set(name, type)
  }

  checkVariable(name) {
    if (!this.symbols.has(name)) throw new Error(`Variable ${name} not declared`)
    return this.symbols.get(name)
  }

  analyze(node) {
    switch (node.type) {
      case 'Literal': return typeof node.value
      case 'Variable': return this.checkVariable(node.name)
      case 'BinaryExpression':
        const leftType = this.analyze(node.left)
        const rightType = this.analyze(node.right)
        if (leftType !== rightType) throw new TypeError(`Type mismatch: ${leftType} vs ${rightType}`)
        return leftType
      default: throw new Error(`Unknown node type: ${node.type}`)
    }
  }
}

// runtime-parser.js
import { Parser } from './parser.js'
import { SemanticAnalyzer } from './semantic.js'
import { MindLangAI } from './mindlang.js'

export class MindLangFullParser {
  constructor(provider) {
    this.ai = new MindLangAI(provider)
  }

  async analyze(source) {
    const parser = new Parser(source)
    let ast
    try {
      ast = parser.parse()
    } catch (err) {
      return { error: `Parse error: ${err.message}` }
    }

    const sem = new SemanticAnalyzer()
    // 예제: 모든 identifier는 number로 선언
    function declareAll(node) {
      if (node.type === 'Variable') sem.declareVariable(node.name, 'number')
      if (node.left) declareAll(node.left)
      if (node.right) declareAll(node.right)
    }
    declareAll(ast)

    let semError = null
    try {
      sem.analyze(ast)
    } catch (err) {
      semError = err.message
    }

    const aiPrompt = `
MindLang AI:
Input AST: ${JSON.stringify(ast, null, 2)}
Semantic Check: ${semError || 'OK'}
Please suggest improvements, comments, or refactor.
`

    const aiResult = await this.ai.run(aiPrompt)

    return {
      ast,
      semanticError: semError,
      ai: aiResult.text
    }
  }
}
