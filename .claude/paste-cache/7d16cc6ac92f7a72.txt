🛠️ Stage 1 Phase 1: Social Media Distributor 설계
단순한 포스팅 도구가 아니라, 각 채널의 **문법(Grammar)**과 **타이밍(Timing)**을 이해하는 지능형 배포 엔진으로 설계합니다.
1. 채널별 어댑터 엔진 (Multi-Channel Adapter)
동일한 blogContext라도 채널의 성격에 따라 데이터의 형태를 재조합합니다.
 * Twitter: 280자 제한 + 핵심 해시태그 + delta.fl 분석 기반의 '한 줄 요약'.
 * LinkedIn: 전문적인 어조 + '설계 결정(Design Decision)' 강조 + 연결된 프로젝트 태깅.
 * Blog: 전체 마크다운 + 코드 스니펫 + tree.fl 시각화 링크.
 * Reddit: 하위 게시판(r/programming 등)의 커뮤니티 가이드라인 준수 여부 체크.
2. 배포 로직: distributor.fl 핵심 스펙
# distributor.fl

fn formatForChannel(context: map, channel: string): map {
    if channel == "twitter" {
        return buildTwitterThread(context) # 스레드 형식으로 분할
    } else if channel == "linkedin" {
        return buildProfessionalPost(context)
    }
    ...
}

fn distributeBlogPost(blogContext: map, socialSummary: map, schedule: i32): map {
    let distributionPlan = {
        "id": generateUUID(),
        "origin_event": blogContext["id"],
        "tasks": []
    }
    
    # 1. 채널별 포맷 변환
    for channel in ["twitter", "linkedin", "blog", "reddit"] {
        let payload = formatForChannel(blogContext, channel)
        distributionPlan["tasks"].append({
            "channel": channel,
            "payload": payload,
            "status": "queued",
            "execute_at": getCurrentTimestamp() + schedule
        })
    }
    
    # 2. 배포 계획을 agent-protocol에 기록 (증명)
    recordEvent("DISTRIBUTION_PLAN_CREATED", distributionPlan)
    
    return distributionPlan
}
