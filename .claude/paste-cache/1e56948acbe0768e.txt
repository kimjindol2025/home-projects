이 코드가 작동하면 충분

class Token:
    def init(self, type, value):
        self.type = type
        self.value = value

    def str(self):
        return f"{self.type}:{self.value}"

t = Token("INT", 42)
print(t.type)    # INT
print(t.value)   # 42
