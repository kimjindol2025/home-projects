rn 0;
        if field == "y" return 4;
    }
    return -1;
}
```

### 4. Code Generation – struct 지원 (x86-64)

`generate_elf()` 내 IR 처리 부분에 추가:

```freelang
for instr in ir_instructions {
    match instr.op {
        STRUCT_CREATE => {
            // sub rsp, size (스택에 struct 할당)
            elf_write(code_pos, "\x48\x83\xec");
            elf_write32(code_pos+3, instr.arg2); // size
            code_pos += 7;
        }

        STRUCT_FIELD_SET => {
            // mov [rsp + offset], val_reg (예: rsi)
            elf_write(code_pos, "\x48\x89\x74\x24");
            elf_write8(code_pos+4, instr.offset);
            code_pos += 5;
        }

        STRUCT_FIELD_GET => {
            // mov rax, [rsp + offset]
            elf_write(code_pos, "\x48\x8b\x44\x24");
            elf_write8(code_pos+4, instr.offset);
            code_pos += 5;
        }

        // 기존 syscall, 배열 등 처리 유지
    }
}
