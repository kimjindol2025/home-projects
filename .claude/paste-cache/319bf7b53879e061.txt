```bash

# 한 번에 두 파일만 실행해서 빠르게 확인
npm test -- src/db-100m-jest.test.ts src/ir-gen-jest.test.ts
```

#### 2. checker.ts 타입 에러 케이스 집중 테스트 예시
```ts
// checker-jest.test.ts 에 추가할 수 있는 예시
describe('Checker - struct 타입 검사', () => {
  it('struct 필드 타입 불일치 시 에러', () => {
    const code = `
      struct Point { x: i32, y: i32 }
      let p: Point = { x: 10, y: "hello" };  // 에러 예상
    `;
    expect(() => check(code)).toThrow(/type mismatch/);
  });

  it('struct 필드 누락 시 에러', () => {
    const code = `
      struct Point { x: i32, y: i32 }
      let p: Point = { x: 10 };  // y 누락
    `;
    expect(() => check(code)).toThrow(/missing field 'y'/);
  });
});
```
