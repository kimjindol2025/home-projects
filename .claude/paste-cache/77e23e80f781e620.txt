 offset = get_field_offset(field);
            let dest_reg = self.alloc_reg();
            self.emit(IRNode{op: STRUCT_FIELD_GET, arg1: struct_reg, arg2: 0, offset: offset, str_idx: dest_reg});
            self.push_result(dest_reg);
        },

        // ... 기존 배열, 기본 expr 처리 ...
    }
}
```

### 3. Code Generation – struct → 메모리 접근 (간단 예시)

`generate_elf()` 함수 안에 struct 처리 추가:

```freelang
// IR → x86-64 변환 (struct 부분)
for instr in ir_instructions {
    match instr.op {
        STRUCT_CREATE => {
            // call alloc (또는 고정 크기 스택 할당)
            elf_write(code_pos, "\x48\x83\xec"); // sub rsp, size
            elf_write32(code_pos+3, 32);         // 예: 32바이트 struct
            code_pos += 7;
        }
        STRUCT_FIELD_SET => {
            // mov [reg + offset], val_reg
            elf_write(code_pos, "\x48\x89");     // mov [rdi + offset], rsi 예시
            elf_write8(code_pos+2, 0x77);        // ModRM
            elf_write32(code_pos+3, instr.offset);
            code_pos += 7;
        }
        STRUCT_FIELD_GET => {
            // mov rax, [reg + offset]
            elf_write(code_pos, "\x48\x8b\x47");
            elf_write8(code_pos+3, instr.offset);
            code_pos += 4;
        }
        // ... 기존 CREATE_ARRAY 등 ...
    }
}
```
