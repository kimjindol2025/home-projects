module ir_v1_kernel {

    import kernel_core

    // ----------------------------
    // Intermediate Representation (IR v1.0) 기본 구조
    // ----------------------------
    type IR_Instruction {
        opcode: string
        operands: list
        metadata: map
    }

    var ir_program: list[IR_Instruction] = []

    func add_ir_instruction(opcode: string, operands: list, metadata: map = {}) {
        var inst = IR_Instruction{opcode=opcode, operands=operands, metadata=metadata}
        ir_program.push(inst)
    }

    // ----------------------------
    // IR -> Task 변환 (Kernel 연동)
    // ----------------------------
    func ir_to_task(ir_inst: IR_Instruction) -> kernel_core.Task {
        var task_id = ir_program.index_of(ir_inst)
        var context = map{"ir_inst": ir_inst}
        var t = kernel_core.create_task(task_id, "IR_Task_" + str(task_id), priority=1, init_context=context)
        return t
    }

    func compile_ir_to_tasks() {
        for inst in ir_program {
            ir_to_task(inst)
        }
    }

    // ----------------------------
    // IR Scheduler (Kernel 스케줄러 호출)
    // ----------------------------
    func run_ir_program() {
        compile_ir_to_tasks()
        while kernel_core.ready_queue.length > 0 {
            kernel_core.schedule()
        }
    }

    // ----------------------------
    // IR Debug
    // ----------------------------
    func print_ir_program() {
        for i, inst in ir_program {
            print("IR", i, "Opcode:", inst.opcode, "Operands:", inst.operands, "Metadata:", inst.metadata)
        }
    }
}

🔹 핵심 내용

1. IR_Instruction 구조체: opcode, operands, metadata 포함


2. IR -> Kernel Task 변환: 각 IR 명령어를 Task로 등록


3. Scheduler 연동: 기존 kernel_core.schedule() 사용


4. Debug / 출력 기능: IR 프로그램 확인 가능
