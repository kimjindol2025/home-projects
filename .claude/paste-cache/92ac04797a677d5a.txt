4. 글로벌 집계 & 배포



func aggregateWeights(nodes: Node[]) -> WeightArray {
    var size = nodes[0].weights.length
    var global: WeightArray = Array(size).fill(0.0)

    for n in nodes {
        for i in 0..size-1 {
            global[i] = global[i] + n.weights[i]
        }
    }

    for i in 0..size-1 {
        global[i] = global[i] / nodes.length
    }

    return global
}

func distributeWeights(nodes: Node[], globalWeights: WeightArray) -> Node[] {
    var updatedNodes: Node[] = []
    for n in nodes {
        updatedNodes.push(updateNode(n, globalWeights))
    }
    return updatedNodes
}

5. 파서 + Executor 예시



func parseCommand(cmd: string) -> WeightArray {
    // 예: "update 0.1 0.2 0.3"
    var tokens = cmd.split(" ")
    var delta: WeightArray = []
    for t in tokens[1..] {
        delta.push(toFloat(t))
    }
    return delta
}

func execute(node: Node, delta: WeightArray) -> Node {
    return updateNode(node, delta)
}
