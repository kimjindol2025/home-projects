### Phase 2 목표 요약 (Minimal Statement / Expression Parser)

**핵심 슬로건**: "간단한 프로그램 하나를 완전히 파싱해서 AST를 뱉을 수 있게 만들기"  
(아직 precedence climbing 전체는 안 해도 됩니다. 최소한의 표현식 + 문장 몇 개만)

**예상 기간**: 4~10일 (하루 2~4시간 투자 기준)  
**난이도**: ★★★☆☆ (lexer보다 논리적으로 복잡하지만, 재귀 구조라 패턴 반복됨)

### Phase 2 추천 구현 순서 (우선순위 높은 순)

| 순서 | 기능 / 함수 이름                  | 왜 먼저? (중요도)                          | 예상 난이도 | 검증 대상 (golden 파일 중)     | 팁 / 주의점 |
|------|-----------------------------------|---------------------------------------------|-------------|--------------------------------|-------------|
| 1    | parseProgram() / parse()          | 전체 진입점                                 | ★★☆☆☆      | 모든 파일                      | statements 배열 수집 |
| 2    | parseStatement()                  | 문장 분기 (if/let/fn/return 등)             | ★★★☆☆      | hello.fl, conditions.fl        | switch-like 분기 |
| 3    | parseVariableDeclaration()        | let x = expr; / const y = expr;             | ★★☆☆☆      | arithmetic.fl                  | isConst 플래그 |
| 4    | parseExpression()                 | expr의 최상위 (precedence 최소 1단계)       | ★★★★☆      | arithmetic.fl                  | binary부터 시작 |
| 5    | parseLiteral()                    | number, string, bool, null                  | ★☆☆☆☆      | 모든 파일                      | 간단 |
| 6    | parseIdentifier() + parseCall()   | foo / foo() / obj.field                     | ★★★☆☆      | functions.fl                   | member expr 먼저 |
| 7    | parseBinaryExpression() (기본)    | + - * / 만 먼저                             | ★★★★☆      | arithmetic.fl                  | left-assoc, no prec table yet |
| 8    | parseBlockStatement()             | { ... }                                     | ★★☆☆☆      | conditions.fl, loops.fl        | statements 반복 |
| 9    | parseIfStatement()                | if cond { } else { }                        | ★★★☆☆      | conditions.fl                  | else optional |
| 10   | parseReturnStatement()            | return expr;                                | ★★☆☆☆      | functions.fl                   | expr optional |

**첫 3~4일 집중 추천**: 1~6번까지 완료  
→ hello.fl + arithmetic.fl 정도는 완전히 AST로 나와야 함

### Phase 2 시작 코드 스켈레톤 (parser.fl에 바로 붙여넣기 추천)

```fl
# --- Parser State ---
let pos = 0          # current token index
let tokens = []      # lexer 결과
let current_token() { tokens[pos] or {type: "EOF"} }

fn advance() {
  pos = pos + 1
}

fn consume(expected_type) {
  let tok = current_token()
  if tok.type != expected_type {
    error("Expected " + expected_type + ", got " + tok.type)
  }
  advance()
  return tok
}

fn peek() { current_token() }

fn error(msg) {
  let tok = current_token()
  println("Parse error at line " + str(tok.line) + ":" + str(tok.col) + " - " + msg)
  # panic이나 recovery 로직 나중에
}

# --- Entry Point ---
fn parse(source) {
  tokens = tokenize(source)
  pos = 0
  return parseProgram()
}

fn parseProgram() {
  let stmts = []
  while peek().type != "EOF" {
    let stmt = parseStatement()
    if stmt != null {
      push(stmts, stmt)
    }
  }
  return node("Program", {body: stmts})
}

# --- Statements ---
fn parseStatement() {
  let tok = peek()
  if tok.type == "LET" or tok.type == "CONST" {
    return parseVariableDeclaration()
  }
  if tok.type == "FN" {
    return parseFunctionDeclaration()
  }
  if tok.type == "IF" {
    return parseIfStatement()
  }
  if tok.type == "RETURN" {
    return parseReturnStatement()
  }
  if tok.type == "LBRACE" {
    return parseBlockStatement()
  }
  # expression statement fallback
  let expr = parseExpression()
  consume("SEMICOLON")
  return node("ExpressionStatement", {expression: expr})
}

# --- 변수 선언 예시 ---
fn parseVariableDeclaration() {
  let is_const = peek().type == "CONST"
  consume(if is_const "CONST" else "LET")
  
  let name = consume("IDENT").value
  
  let init = null
  if peek().type == "ASSIGN" {
    consume("ASSIGN")
    init = parseExpression()
  }
  
  consume("SEMICOLON")
  
  return node("VariableDeclaration", {
    kind: if is_const "const" else "let",
    name: name,
    init: init
  })
}

# --- Expression 기본 (나중에 확장) ---
fn parseExpression() {
  return parseLiteral() or parseIdentifier() or parseBinaryExpression()
}

fn parseLiteral() {
  let tok = peek()
  if tok.type == "NUMBER" {
    advance()
    return node("Literal", {value: tok.value, raw: tok.value})
  }
  if tok.type == "STRING" {
    advance()
    return node("Literal", {value: tok.value, raw: tok.value})
  }
  # bool, null 등 추가
  return null
}

# ... 나머지 함수들 계속 추가
```
