2️⃣ VM에 제어 흐름 추가

// freelang-vm.ts (switch 내부에 추가)

case OpCode.JMP:
  this.ip = instr.arg!
  continue

case OpCode.JZ: {
  const value = this.stack.pop()!
  if (value === 0) {
    this.ip = instr.arg!
    continue
  }
  break
}

case OpCode.JNZ: {
  const value = this.stack.pop()!
  if (value !== 0) {
    this.ip = instr.arg!
    continue
  }
  break
}

중요:

continue 사용해야 ip++ 방지됨

점프 후 즉시 새 위치 실행
