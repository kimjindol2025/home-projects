module ir_v1_runtime_optimization {

    import kernel_core
    import ir_v1_kernel

    // ----------------------------
    // Runtime 최적화 옵션
    // ----------------------------
    var OPT_INLINE_THRESHOLD: int = 3      // 작은 함수 인라인
    var OPT_CONSTANT_FOLDING: bool = true  // 상수 연산 미리 계산
    var OPT_DEAD_CODE_ELIM: bool = true    // 사용되지 않는 IR 제거

    // ----------------------------
    // IR 최적화 함수
    // ----------------------------
    func optimize_ir_program() {
        var new_ir: list[ir_v1_kernel.IR_Instruction] = []
        for inst in ir_v1_kernel.ir_program {

            // 상수 폴딩
            if OPT_CONSTANT_FOLDING {
                if inst.opcode == "ADD" and is_constant(inst.operands[0]) and is_constant(inst.operands[1]) {
                    var val = inst.operands[0] + inst.operands[1]
                    inst = ir_v1_kernel.IR_Instruction{opcode="LOAD_CONST", operands=[val], metadata=inst.metadata}
                }
            }

            // Dead code 제거
            if OPT_DEAD_CODE_ELIM {
                if inst.opcode == "NOP" {
                    continue
                }
            }

            new_ir.push(inst)
        }
        ir_v1_kernel.ir_program = new_ir
    }

    // ----------------------------
    // Task-level 최적화
    // ----------------------------
    func optimize_tasks() {
        for t in kernel_core.ready_queue {
            // 우선순위 조정: IR 연산에 따른 동적 재배치
            if t.context["ir_inst"].opcode == "CRITICAL" {
                t.priority = t.priority + 1
            }
        }
    }

    // ----------------------------
    // 최적화 실행
    // ----------------------------
    func run_optimized_ir_program() {
        optimize_ir_program()
        ir_v1_kernel.compile_ir_to_tasks()
        optimize_tasks()

        while kernel_core.ready_queue.length > 0 {
            kernel_core.schedule()
        }
    }

    // ----------------------------
    // Debug: 최적화 전/후 IR 출력
    // ----------------------------
    func debug_ir_optimization() {
        print("=== Before Optimization ===")
        ir_v1_kernel.print_ir_program()
        optimize_ir_program()
        print("=== After Optimization ===")
        ir_v1_kernel.print_ir_program()
    }
}

🔹 핵심 내용

1. 상수 폴딩(Constant Folding): 계산 가능한 상수 연산을 컴파일 시 미리 계산


2. Dead Code Elimination: NOP나 사용되지 않는 IR 제거


3. Task 우선순위 최적화: 중요한 IR 명령은 우선 실행


4. Debug 기능: 최적화 전후 IR 비교 가능
