🛠️ Stage 1: FreeLang storage.fl 실전 구현
추상화된 스텁(Stub)을 걷어내고, 실제 파일 시스템에 .freegit 구조를 생성하는 핵심 로직입니다.
1. 디렉토리 재귀 생성 및 객체 경로 계산
Git의 성능 비결인 objects/xx/yyyy... 구조를 프리랭의 문자열 제어 함수로 구현합니다.
# storage.fl (실전용)

# 객체 저장 경로 생성 로직
fn getObjectPath(repoPath: string, hash: string): string {
    let dirPart = substring(hash, 0, 2)
    let filePart = substring(hash, 2, len(hash))
    return repoPath + "/.freegit/objects/" + dirPart + "/" + filePart
}

# 실제 파일 쓰기 (기존 stub 대체)
fn storeObject(repoPath: string, hash: string, fullData: string): bool {
    let path = getObjectPath(repoPath, hash)
    let dir = repoPath + "/.freegit/objects/" + substring(hash, 0, 2)
    
    # 1. 디렉토리가 없으면 생성 (native 호출 가정)
    if !exists(dir) {
        createDirectory(dir)
    }
    
    # 2. 파일 쓰기
    return writeFileContent(path, fullData)
}
