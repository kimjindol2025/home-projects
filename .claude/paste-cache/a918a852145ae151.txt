 // scanCodebase 코드 파일 수집
 func scanCodebase() []FileInfo {
   var files []FileInfo

   for _, root := range scanRoots {
     if _, err := os.Stat(root); err != nil {
       continue
     }
     filepath.Walk(root, func(path string, info
 os.FileInfo, err error) error {
       if err != nil || info.IsDir() {
         return nil
       }
       // 불필요 디렉토리 스킵
       for _, skip := range []string{"target",
 "node_modules", ".git", "dist"} {
         if strings.Contains(path, "/"+skip+"/") {
           return nil
         }
       }
       ext := strings.ToLower(filepath.Ext(path))
       lang, ok := codeExts[ext]
       if !ok {
         return nil
       }
       data, err := os.ReadFile(path)
       if err != nil {
         return nil
       }
       content := string(data)
       lines := strings.Count(content, "\n") + 1

       // 너무 작은 파일 스킵 (5줄 미만)
       if lines < 5 {
         return nil
       }
       // 너무 큰 파일은 앞부분만
       if len(content) > 8000 {
         content = content[:8000]
       }

       rel := path
       if idx := strings.Index(path, "/.projects/");
 idx >= 0 {
         rel = path[idx+1:]
       } else if idx := strings.Index(path,
 "/projects/"); idx >= 0 {
         rel = path[idx+1:]
       }

       files = append(files, FileInfo{
         Path:    path,
         RelPath: rel,
         Ext:     ext,
         Lines:   lines,
         Content: content,
         Lang:    lang,
       })
       return nil
     })
   }
   return files
 }

 // buildQAPairs 파일 목록 → QA 쌍 생성
 func buildQAPairs(files []FileInfo) []QAPair {
   var pairs []QAPair

   // 프로젝트별 파일 그룹화
   projectFiles := map[string][]FileInfo{}
   for _, f := range files {
     proj := extractProject(f.RelPath)
     projectFiles[proj] = append(projectFiles[proj],
 f)
   }

   // 각 파일에서 QA 추출
   for _, f := range files {
     pairs = append(pairs, fileQA(f)...)
   }

   // 프로젝트 단위 QA
   for proj, pfiles := range projectFiles {
     pairs = append(pairs, projectQA(proj,
 pfiles)...)
   }

   // 전체 코드베이스 QA
   pairs = append(pairs, globalQA(files)...)

   return pairs
 }

 // fileQA 단일 파일에서 QA 생성
 func fileQA(f FileInfo) []QAPair {
   var pairs []QAPair
   name := filepath.Base(f.Path)
   proj := extractProject(f.RelPath)

   // 1. 파일 존재 여부
   pairs = append(pairs, QAPair{
     Question: fmt.Sprintf("%s 파일이 있어?", name),
     Answer:   fmt.Sprintf("있음. %s (%d줄, %s)",
 f.RelPath, f.Lines, f.Lang),
     Source:   f.RelPath,
   })

   // 2. 파일 역할 (이름 기반 추론)
   role := inferFileRole(name, f.Content)
   if role != "" {
     pairs = append(pairs, QAPair{
       Question: fmt.Sprintf("%s에서 %s 어디 있어?",
 proj, role),
       Answer:   fmt.Sprintf("%s (%d줄)", f.RelPath,
 f.Lines),
       Source:   f.RelPath,
     })
   }

   // 3. 코드 내 주요 구조체/함수 추출
   structs := extractStructs(f.Content, f.Lang)
   for _, s := range structs {
     pairs = append(pairs, QAPair{
       Question: fmt.Sprintf("%s가 뭐야?", s),
       Answer:   fmt.Sprintf("%s에 정의된
 구조체/타입. 파일: %s", s, f.RelPath),
       Source:   f.RelPath,
     })
   }

   funcs := extractFunctions(f.Content, f.Lang)
   for _, fn := range funcs[:min(len(funcs), 3)] {
     pairs = append(pairs, QAPair{
       Question: fmt.Sprintf("%s 함수 어디 있어?",
 fn),
       Answer:   fmt.Sprintf("%s에 구현됨 (%s)",
 f.RelPath, f.Lang),
       Source:   f.RelPath,
     })
   }

   return pairs
 }

 // projectQA 프로젝트 단위 QA
 func projectQA(proj string, files []FileInfo)
 []QAPair {
   var pairs []QAPair
   if len(files) == 0 {
     return pairs
   }

   // 파일 목록
   var names []string
   totalLines := 0
   langs := map[string]int{}
   for _, f := range files {
     names = append(names, filepath.Base(f.Path))
     totalLines += f.Lines
     langs[f.Lang]++
   }

   langStr := ""
   for l, c := range langs {
     langStr += fmt.Sprintf("%s(%d) ", l, c)
   }

   pairs = append(pairs, QAPair{
     Question: fmt.Sprintf("%s 프로젝트에 뭐가
 있어?", proj),
     Answer:   fmt.Sprintf("%d개 파일, 총 %d줄. 언어:
  %s. 파일: %s", len(files), totalLines,
 strings.TrimSpace(langStr),
 strings.Join(names[:min(len(names), 8)], ", ")),
     Source:   proj,
   })

   pairs = append(pairs, QAPair{
     Question: fmt.Sprintf("%s 코드 몇 줄이야?",
 proj),
     Answer:   fmt.Sprintf("총 %d줄 (%d개 파일)",
 totalLines, len(files)),
     Source:   proj,
   })

   return pairs
 }

 // globalQA 전체 코드베이스 QA
 func globalQA(files []FileInfo) []QAPair {
   var pairs []QAPair

   // GC 관련
   gcFiles := filterByKeyword(files, []string{"gc",
 "garbage", "mark_sweep", "generational"})
   if len(gcFiles) > 0 {
     var desc []string
     for _, f := range gcFiles {
       desc = append(desc, fmt.Sprintf("%s(%d줄)",
 filepath.Base(f.Path), f.Lines))
     }
     ans := fmt.Sprintf("있음. %d개 파일: %s",
 len(gcFiles), strings.Join(desc, ", "))
     for _, q := range []string{"FreeLang에 GC
 있어?", "가비지 컬렉션 구현됐어?", "메모리 관리
 어떻게 해?"} {
       pairs = append(pairs, QAPair{Question: q,
 Answer: ans, Source: "global"})
     }
   }

   // JIT 관련
   jitFiles := filterByKeyword(files, []string{"jit",
  "hotspot", "native-codegen", "inline-cache"})
   if len(jitFiles) > 0 {
     var desc []string
     for _, f := range jitFiles {
       desc = append(desc, fmt.Sprintf("%s(%d줄)",
 filepath.Base(f.Path), f.Lines))
     }
     ans := fmt.Sprintf("있음. %d개 파일: %s",
 len(jitFiles), strings.Join(desc, ", "))
     for _, q := range []string{"JIT 컴파일러 있어?",
  "FreeLang JIT 어디 있어?", "핫스팟 최적화 있어?"} {
       pairs = append(pairs, QAPair{Question: q,
 Answer: ans, Source: "global"})
     }
   }

   // escape analysis
   escFiles := filterByKeyword(files,
 []string{"escape_analysis", "escape analysis",
 "EscapeAnalysis"})
   escAns := "없음. 현재 미구현."
   if len(escFiles) > 0 {
     escAns = fmt.Sprintf("있음. %d개 파일에서 발견",
  len(escFiles))
   }
   pairs = append(pairs, QAPair{Question: "escape
 analysis 있어?", Answer: escAns, Source: "global"})
   pairs = append(pairs, QAPair{Question: "FreeLang에
  escape analysis 구현됐어?", Answer: escAns, Source:
  "global"})

   // 전체 통계
   totalLines := 0
   for _, f := range files {
     totalLines += f.Lines
   }
   pairs = append(pairs, QAPair{
     Question: "FreeLang 코드베이스 총 규모가 어떻게
 돼?",
     Answer:   fmt.Sprintf("총 %d개 파일, %d줄.
 Rust/Go/TypeScript/FreeLang 혼합.", len(files),
 totalLines),
     Source:   "global",
   })

   return pairs
 }

 // filterByKeyword 파일 이름이나 내용에 키워드
 포함된 파일 필터
 func filterByKeyword(files []FileInfo, keywords
 []string) []FileInfo {
   var result []FileInfo
   for _, f := range files {
     nameAndContent := strings.ToLower(f.Path + " " +
  f.Content[:min(len(f.Content), 500)])
     for _, kw := range keywords {
       if strings.Contains(nameAndContent, kw) {
         result = append(result, f)
         break
       }
     }
   }
   return result
 }

 // extractProject 경로에서 프로젝트명 추출
 func extractProject(relPath string) string {
   parts := strings.Split(relPath, "/")
   for i, p := range parts {
     if p == "modules" || p == "core" || p ==
 "projects" {
       if i+1 < len(parts) {
         return parts[i+1]
       }
     }
   }
   if len(parts) > 0 {
     return parts[0]
   }
   return "unknown"
 }

 // extractStructs 코드에서 구조체/타입 이름 추출
 func extractStructs(content, lang string) []string {
   var result []string
   lines := strings.Split(content, "\n")
   for _, line := range lines {
     line = strings.TrimSpace(line)
     var name string
     switch lang {
     case "rust":
       if strings.HasPrefix(line, "pub struct ") ||
 strings.HasPrefix(line, "struct ") {
         parts := strings.Fields(line)
         if len(parts) >= 2 {
           name = strings.TrimSuffix(parts[1], "{")
         }
       }
     case "go":
       if strings.HasPrefix(line, "type ") &&
 strings.Contains(line, "struct") {
         parts := strings.Fields(line)
         if len(parts) >= 2 {
           name = parts[1]
         }
       }
     case "typescript":
       if strings.HasPrefix(line, "interface ") ||
 strings.HasPrefix(line, "class ") ||
 strings.HasPrefix(line, "type ") {
         parts := strings.Fields(line)
         if len(parts) >= 2 {
           name = strings.TrimSuffix(parts[1], "{")
         }
       }
     }
     if name != "" && len(name) > 2 && len(name) < 40
  {
       result = append(result, name)
     }
   }
   return result
 }

 // extractFunctions 코드에서 함수 이름 추출
 func extractFunctions(content, lang string) []string
  {
   var result []string
   lines := strings.Split(content, "\n")
   for _, line := range lines {
     line = strings.TrimSpace(line)
     var name string
     switch lang {
     case "rust":
       if strings.HasPrefix(line, "pub fn ") ||
 strings.HasPrefix(line, "fn ") {
         parts := strings.Fields(line)
         if len(parts) >= 2 {
           fn := parts[1]
           if idx := strings.Index(fn, "("); idx > 0
 {
             name = fn[:idx]
           }
         }
       }
     case "go":
       if strings.HasPrefix(line, "func ") {
         parts := strings.Fields(line)
         if len(parts) >= 2 {
           fn := parts[1]
           if idx := strings.Index(fn, "("); idx > 0
 {
             name = fn[:idx]
           }
         }
       }
     }
     if name != "" && len(name) > 2 && len(name) < 40
  {
       result = append(result, name)
     }
   }
   return result
 }

 func min(a, b int) int {
   if a < b {
     return a
   }
   return b
 }

 // saveQA JSONL 형식으로 저장
 func saveQA(pairs []QAPair, path string) {
   f, err := os.Create(path)
   if err != nil {
     fmt.Printf("❌ 저장 실패: %v\n", err)
     return
   }
   defer f.Close()

   enc := json.NewEncoder(f)
   for _, p := range pairs {
     enc.Encode(p)
   }
 }
