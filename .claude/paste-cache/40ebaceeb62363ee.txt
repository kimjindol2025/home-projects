1️⃣ Node.js/JS 기반 마이크로커널 시뮬레이션

// microkernel.js
class Process {
  constructor(pid, name, code) {
    this.pid = pid
    this.name = name
    this.code = code
    this.state = 'ready' // ready, running, waiting, terminated
    this.pc = 0
  }
}

class Kernel {
  constructor() {
    this.processes = []
    this.pidCounter = 0
  }

  createProcess(name, code) {
    const pid = ++this.pidCounter
    const proc = new Process(pid, name, code)
    this.processes.push(proc)
    return proc
  }

  schedule() {
    // Round-robin 스케줄링 예제
    for (const proc of this.processes) {
      if (proc.state === 'ready') {
        proc.state = 'running'
        try {
          const instr = proc.code[proc.pc++]
          if (instr) instr()
        } catch (e) {
          console.error(`Process ${proc.pid} crashed: ${e.message}`)
        }
        proc.state = 'ready'
      }
    }
  }

  run(ticks = 10) {
    for (let i = 0; i < ticks; i++) this.schedule()
  }
}

// 예제
const kernel = new Kernel()
kernel.createProcess('proc1', [
  () => console.log('proc1 step1'),
  () => console.log('proc1 step2')
])
kernel.createProcess('proc2', [
  () => console.log('proc2 step1')
])

kernel.run()

2️⃣ Rust/C 기반 실제 OS 커널 설계 방향

1. 부트스트랩

bootloader → CPU 초기화 → 메모리 초기화



2. 메모리 관리

페이지 테이블, 프리 리스트, 커널/유저 영역 분리



3. 프로세스 관리

PCB(Process Control Block), 컨텍스트 스위치



4. 스케줄러

Round-robin / Priority / Multi-level queue



5. 시스템 콜

입출력, 파일, IPC



6. 인터럽트 처리

타이머, I/O, 소프트웨어 인터럽트



7. 파일 시스템

FAT/EXT2 간단 구현



8. 디바이스 드라이버

UART, 키보드, 디스크 시뮬레이션

예시: Rust 최소 커널 (no_std)

#![no_std]
#![no_main]

use core::panic::PanicInfo;

#[no_mangle]
pub extern "C" fn _start() -> ! {
    kernel_main();
}

fn kernel_main() -> ! {
    println!("Hello, Microkernel!");
    loop {}
}

#[panic_handler]
fn panic(_info: &PanicInfo) -> ! {
    loop {}
}

no_std → 표준 라이브러리 없이 커널 레벨

_start → 부트로더에서 진입점

실제 OS는 여기서 메모리 초기화, GDT, IDT, 인터럽트 등을 추가
